"""Clone an actual native past for one current-token causal measurement.

The clone is ephemeral. It is not a reusable past for a changed conversation.
Only the pinned DynamicCache with initialized dense KV layers is accepted.
"""

import copy
import hashlib
import json
from contextlib import contextmanager

import torch
from transformers.cache_utils import DynamicCache, DynamicLayer, DynamicSlidingWindowLayer


def tensor_hash(value):
    return hashlib.sha256(value.detach().contiguous().reshape(-1).view(torch.uint8).cpu().numpy().tobytes()).hexdigest()


def _value(value):
    if torch.is_tensor(value):
        return {'shape': list(value.shape), 'dtype': str(value.dtype), 'device': str(value.device), 'sha256': tensor_hash(value)}
    if value is None or type(value) in (str, bool, int, float):
        return value
    if isinstance(value, (torch.dtype, torch.device)):
        return str(value)
    raise ValueError('Unsupported mutable cache metadata')


def fingerprint(cache, entries, length, dtype):
    """Validate logical and physical layout and retain exact compact witnesses."""
    if type(cache) is not DynamicCache or cache.offloading or cache.layer_class_to_replicate is not None:
        raise ValueError('Initialized non-offloaded DynamicCache required')
    if type(entries) is not int or type(length) is not int or min(entries, length) < 1 or len(cache.layers) != entries:
        raise ValueError('Exact initialized cache cardinality and length required')
    if set(cache.__dict__) != {'layers', 'layer_class_to_replicate', 'offloading'}:
        raise ValueError('Unexpected cache implementation state')
    rows = []
    for index, layer in enumerate(cache.layers):
        if type(layer) not in (DynamicLayer, DynamicSlidingWindowLayer) or not layer.is_initialized:
            raise ValueError('Actual initialized native layer required')
        if int(layer.get_seq_length()) != length:
            raise ValueError('Logical cache length differs from declared native past')
        k, v = layer.keys, layer.values
        if not torch.is_tensor(k) or not torch.is_tensor(v) or k.shape != v.shape or k.ndim != 4 or k.shape[0] != 1:
            raise ValueError('Dense batch-one aligned key/value tensors required')
        physical = min(length, layer.sliding_window - 1) if type(layer) is DynamicSlidingWindowLayer else length
        if k.shape[-2] != physical or min(k.shape) < 1 or k.dtype != dtype or v.dtype != dtype or k.device != v.device:
            raise ValueError('Truncated, mismatched or wrong-dtype native cache')
        if not torch.isfinite(k).all() or not torch.isfinite(v).all():
            raise ValueError('Nonfinite native cache')
        rows.append({'layer': index, 'type': type(layer).__name__, 'attributes': {key: _value(value) for key, value in sorted(layer.__dict__.items())}})
    raw = json.dumps(rows, sort_keys=True, separators=(',', ':'))
    return {'entries': entries, 'logical_length': length, 'layers': rows, 'sha256': hashlib.sha256(raw.encode()).hexdigest()}


def _storages(cache):
    return {value.untyped_storage().data_ptr() for layer in cache.layers for value in layer.__dict__.values() if torch.is_tensor(value) and value.numel()}


@contextmanager
def native_cache_branch(cache, entries, length, dtype):
    """Yield a fresh nonaliasing native past and require one-token consumption.

    The source must stay byte-exact, including sliding-window counters. A failed
    probe still checks source immutability. Full raw pasts are never serialized.
    """
    before = fingerprint(cache, entries, length, dtype)
    branch = copy.deepcopy(cache)
    if fingerprint(branch, entries, length, dtype) != before or _storages(cache) & _storages(branch):
        raise ValueError('Cache copy is inexact or aliases native storage')
    success = False
    try:
        yield branch
        success = True
    finally:
        if fingerprint(cache, entries, length, dtype) != before:
            raise AssertionError('Native source past changed during measurement')
        if success:
            fingerprint(branch, entries, length + 1, dtype)

"""Read native MLP activity without changing its forward tensors or parameters."""

import torch


def spaced_positions(length, maximum):
    """Choose deterministic integer endpoints without response-dependent selection."""
    if not isinstance(length, int) or not isinstance(maximum, int):
        raise ValueError('Integer lengths required')
    if min(length, maximum) < 1:
        raise ValueError('Positive lengths required')
    count = min(length, maximum)
    return [0] if count == 1 else [i * (length - 1) // (count - 1) for i in range(count)]


class NativeActivity:
    """Measure post-gating channels and both sides of post-MLP normalization.

    This observer retains selected token slices and streaming moments. It never
    returns a replacement forward value. Call ``begin`` before each model call
    and ``finish`` afterward. Moments cover only the last processed position of
    each call, avoiding extra weight for long prompt prefills.
    """

    def __init__(self, model, coordinates):
        self.layers = list(model.model.language_model.layers)
        self.coordinates = [tuple(x) for x in coordinates]
        if not self.coordinates or self.coordinates != sorted(set(self.coordinates)):
            raise ValueError('Coordinates must be nonempty, unique and sorted')
        self.widths = [layer.mlp.down_proj.in_features for layer in self.layers]
        for layer, channel in self.coordinates:
            if type(layer) is not int or type(channel) is not int:
                raise ValueError('Integer coordinates required')
            if not 0 <= layer < len(self.layers) or not 0 <= channel < self.widths[layer]:
                raise ValueError('Coordinate outside native MLP')
        self.indices = {
            layer: [channel for li, channel in self.coordinates if li == layer]
            for layer in range(len(self.layers))
        }
        with torch.no_grad():
            self.down_norms = [layer.mlp.down_proj.weight.detach().float().norm(dim=0) for layer in self.layers]
        if not all(torch.isfinite(x).all() for x in self.down_norms):
            raise ValueError('Nonfinite native output columns')
        self.handles = []
        self.active = None
        self.reset_moments()

    def reset_moments(self):
        if self.active is not None:
            raise ValueError('Cannot reset an unfinished forward')
        self.steps = 0
        self.sums = [torch.zeros((3, w), dtype=torch.float64, device=n.device) for w, n in zip(self.widths, self.down_norms)]

    def begin(self, sequence_length, positions, toplist=False):
        if not self.handles or self.active is not None:
            raise ValueError('Observer is closed or a forward is unfinished')
        if type(sequence_length) is not int or sequence_length < 1:
            raise ValueError('Positive native sequence length required')
        positions = list(positions)
        if positions != sorted(set(positions)) or any(type(i) is not int or not 0 <= i < sequence_length for i in positions):
            raise ValueError('Invalid sampled positions')
        self.active = {'length': sequence_length, 'positions': positions, 'toplist': bool(toplist), 'h': {}, 'raw': {}, 'norm': {}, 'tops': {}}

    def _check(self, value, layer, key, width):
        frame = self.active
        if frame is None or layer in frame[key]:
            raise ValueError('Missing begin or repeated native module call')
        if value.ndim != 3 or tuple(value.shape) != (1, frame['length'], width):
            raise ValueError('Unexpected native tensor shape')
        if not torch.isfinite(value).all():
            raise ValueError('Nonfinite observed tensor')
        return frame

    @torch.no_grad()
    def _input(self, layer, module, args):
        value = args[0].detach()
        frame = self._check(value, layer, 'h', self.widths[layer])
        current = value[0, -1].double()
        self.sums[layer][0].add_(current)
        self.sums[layer][1].add_(current.abs())
        self.sums[layer][2].add_(current.square())
        sample = value[0, frame['positions']].float()
        picked = sample[:, self.indices[layer]]
        stats = torch.stack((sample.mean(-1), sample.square().mean(-1).sqrt(), sample.abs().amax(-1)), dim=-1)
        frame['h'][layer] = (picked.cpu(), stats.cpu())
        if frame['toplist']:
            magnitudes = current.float().abs() * self.down_norms[layer]
            order = torch.argsort(magnitudes, descending=True, stable=True)[:min(8, self.widths[layer])]
            frame['tops'][layer] = {
                'channels': order.cpu().tolist(),
                'activation': current[order].float().cpu().tolist(),
                'raw_projection_magnitude': magnitudes[order].cpu().tolist(),
            }

    @torch.no_grad()
    def _output(self, layer, key, module, args, output):
        value = output.detach()
        width = self.layers[layer].mlp.down_proj.out_features
        frame = self._check(value, layer, key, width)
        frame[key][layer] = value[0, frame['positions']].float().norm(dim=-1).cpu()

    def finish(self):
        frame = self.active
        if frame is None:
            raise ValueError('No forward is active')
        expected = set(range(len(self.layers)))
        if any(set(frame[key]) != expected for key in ['h', 'raw', 'norm']):
            raise ValueError('Incomplete native module coverage')
        if frame['toplist'] and set(frame['tops']) != expected:
            raise ValueError('Incomplete native toplist coverage')
        units = torch.cat([frame['h'][i][0] for i in range(len(self.layers))], dim=1)
        stats = torch.stack([
            torch.cat((frame['h'][i][1], frame['raw'][i][:, None], frame['norm'][i][:, None]), dim=-1)
            for i in range(len(self.layers))
        ], dim=1)
        self.active = None
        self.steps += 1
        return {'values': units.numpy(), 'layer_stats': stats.numpy(), 'tops': frame['tops']}

    def moments(self):
        if self.active is not None or self.steps < 1:
            raise ValueError('No complete trajectory moments')
        return torch.cat(self.sums, dim=1) / self.steps

    def __enter__(self):
        if self.handles:
            raise ValueError('Observer already installed')
        try:
            for i, layer in enumerate(self.layers):
                def pre(module, args, i=i):
                    self._input(i, module, args)
                def raw(module, args, output, i=i):
                    self._output(i, 'raw', module, args, output)
                def norm(module, args, output, i=i):
                    self._output(i, 'norm', module, args, output)
                self.handles.append(layer.mlp.down_proj.register_forward_pre_hook(pre))
                self.handles.append(layer.mlp.down_proj.register_forward_hook(raw))
                self.handles.append(layer.post_feedforward_layernorm.register_forward_hook(norm))
        except BaseException:
            self.__exit__(None, None, None)
            raise
        return self

    def __exit__(self, exc_type, exc, tb):
        unfinished = self.active is not None
        for handle in self.handles:
            handle.remove()
        self.handles = []
        self.active = None
        if unfinished and exc_type is None:
            raise ValueError('Observer exited before a forward finished')

"""PyTorch MLP-channel measurement and reversible interventions.

Use ordinary eager Transformers/PEFT execution. Some optimized kernels bypass
module hooks; finish() fails if any expected hook was not observed.
"""
from contextlib import contextmanager
import re

import torch


def mlp_modules(model):
    candidates={name:module for name,module in model.named_modules()
                if name.endswith('.mlp.down_proj')}
    # Multimodal Gemma also has vision MLPs with the same projection names.
    # Prefer the explicitly named text decoder; text-only models need no prefix.
    text={name:module for name,module in candidates.items()
          if re.search(r'(^|\.)language_model\.layers\.\d+\.mlp\.down_proj$',name)}
    return text if text else candidates


class ActivationRecorder:
    def __init__(self,model):
        self.modules=mlp_modules(model)
        if not self.modules:raise ValueError('No MLP down projections found')
        self.handles=[];self.stats={};self.mask=None

    def set_mask(self,attention_mask):
        self.mask=attention_mask

    def _hook(self,name):
        def capture(module,args):
            values=args[0].detach().float()
            if values.ndim!=3:raise ValueError('Require [batch, tokens, channels] eager activations')
            if self.mask is None or tuple(self.mask.shape)!=tuple(values.shape[:2]):
                raise ValueError('Provide a matching token mask; free-generation traces are not aligned probes')
            selected=values[self.mask.to(device=values.device,dtype=torch.bool)]
            if not len(selected):raise ValueError('Probe contains no unmasked tokens')
            count=selected.shape[0]
            if name not in self.stats:
                self.stats[name]={'count':0,'sum':torch.zeros(values.shape[-1],device=values.device),
                    'sum_abs':torch.zeros(values.shape[-1],device=values.device),
                    'sum_sq':torch.zeros(values.shape[-1],device=values.device)}
            item=self.stats[name];item['count']+=count
            item['sum']+=selected.sum(dim=0)
            item['sum_abs']+=selected.abs().sum(dim=0)
            item['sum_sq']+=(selected*selected).sum(dim=0)
        return capture

    def __enter__(self):
        self.handles=[module.register_forward_pre_hook(self._hook(name)) for name,module in self.modules.items()]
        return self

    def finish(self):
        missing=set(self.modules)-set(self.stats)
        if missing:raise RuntimeError('Unobserved hooks; optimized execution may bypass instrumentation: '+str(sorted(missing)))
        result={}
        for name,item in self.stats.items():
            if any(not torch.isfinite(item[key]).all().item() for key in ('sum','sum_abs','sum_sq')):
                raise RuntimeError('Nonfinite activation statistics at '+name)
            n=item['count'];mean=item['sum']/n
            result[name]={'token_count':n,'mean':mean.cpu(),
                'mean_abs':(item['sum_abs']/n).cpu(),
                'rms':(item['sum_sq']/n).sqrt().cpu(),
                'variance':(item['sum_sq']/n-mean*mean).clamp_min(0).cpu()}
        return result

    def __exit__(self,*args):
        for handle in self.handles:handle.remove()
        self.handles=[]


@contextmanager
def scale_mlp_channels(model,layer,channels,scale):
    """Multiply post-gating MLP channels before down_proj, then restore hooks.

    This does not prune dense matrix multiplies and makes no speed guarantee.
    It affects all token positions during the context's lifetime.
    """
    if not 0<=scale<=2:raise ValueError('Initial intervention range is [0,2]')
    targets=[(name,module) for name,module in mlp_modules(model).items()
             if re.search(rf'layers\.{int(layer)}\.mlp\.down_proj$',name)]
    if len(targets)!=1:raise ValueError('Layer must identify exactly one MLP')
    indices=sorted(set(int(j) for j in channels))
    if not indices or min(indices)<0:raise ValueError('Nonempty nonnegative channel IDs required')
    def alter(module,args):
        source=args[0]
        if max(indices)>=source.shape[-1]:raise ValueError('Channel ID outside layer width')
        changed=source.clone();changed[...,indices]*=scale
        return (changed,*args[1:])
    handle=targets[0][1].register_forward_pre_hook(alter)
    try:yield
    finally:handle.remove()

    """Keep or drop selected LoRA modules while leaving base weights intact."""
    if mode not in ('keep','drop'):raise ValueError('mode must be keep or drop')
    modules={name:module for name,module in model.named_modules()
             if hasattr(module,'lora_A') and callable(getattr(module,'enable_adapters',None))}

"""Describe output text regions without grading or executing a program."""

import ast
import re


def python_regions(text):
    pattern = re.compile(r'```(?:python3?|py)?[ \t]*\r?\n(?P<body>.*?)```', re.I | re.S)
    regions = [list(m.span('body')) for m in pattern.finditer(text)]
    if not regions:
        try:
            tree = ast.parse(text)
        except (SyntaxError, ValueError):
            tree = None
        if tree and any(isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef)) for n in ast.walk(tree)):
            regions = [[0, len(text)]]
    return regions


def label_prefixes(text, prefixes, token_ids, special_ids, coding=True):
    """Unstable Unicode prefix decoding remains explicitly unaligned.

    Labels describe textual regions, not executed computation or neuron roles.
    The caller supplies decoded native prefixes; no generation occurs here.
    """
    if len(prefixes) != len(token_ids) or not prefixes or prefixes[-1] != text:
        raise ValueError('Complete exact native decoding is required')
    regions = python_regions(text) if coding else []
    previous, previous_ok = '', True
    result = []
    for index, (decoded, token) in enumerate(zip(prefixes, token_ids)):
        current_ok = text.startswith(decoded)
        start, end = len(previous), len(decoded)
        aligned = previous_ok and current_ok and start <= end
        if token in special_ids:
            label = 'special'
        elif not aligned:
            label = 'unaligned'
        elif start == end:
            label = 'zero_width'
        elif not coding:
            label = 'retention_content'
        elif any(a <= start and end <= b for a, b in regions):
            label = 'python_region'
        elif any(start < b and end > a for a, b in regions):
            label = 'boundary'
        else:
            label = 'outside_python_region'
        result.append({'generated_index': index, 'category': label, 'start': start if aligned else None, 'end': end if aligned else None})
        previous, previous_ok = decoded, current_ok
    return {'python_regions': regions, 'tokens': result}

"""Label Python text syntax without execution, grading or model inference."""

import io
import keyword
import token
import tokenize


def lexical_regions(text, regions):
    output, errors = [], []
    for start, end in regions:
        body = text[start:end]
        lines = body.splitlines(keepends=True)
        offsets = [0]
        for line in lines:
            offsets.append(offsets[-1] + len(line))
        def absolute(position):
            line, column = position
            return start + offsets[min(line - 1, len(offsets) - 1)] + column
        try:
            for item in tokenize.generate_tokens(io.StringIO(body).readline):
                kind = {
                    token.COMMENT: 'comment', token.STRING: 'string',
                    token.NUMBER: 'number', token.OP: 'operator',
                    token.NAME: 'keyword' if keyword.iskeyword(item.string) else 'name',
                }.get(item.type)
                if kind is not None:
                    a, b = absolute(item.start), absolute(item.end)
                    output.append({'start': a, 'end': b, 'category': kind})
        except (tokenize.TokenError, IndentationError, SyntaxError) as error:
            errors.append({'region_start': start, 'type': type(error).__name__})
    return output, errors


def label_lexical(text, annotation):
    syntax, errors = lexical_regions(text, annotation['python_regions'])
    result = []
    for item in annotation['tokens']:
        category = item['category']
        if category == 'python_region':
            kinds = {s['category'] for s in syntax if item['start'] < s['end'] and item['end'] > s['start']}
            category = next(iter(kinds)) if len(kinds) == 1 else 'mixed_python' if kinds else 'python_whitespace_or_unparsed'
        result.append({**item, 'category': category})
    return {'tokens': result, 'tokenization_errors': errors}

def make_trace(record, values, coordinates, down_norms, provenance, token_label=None):
    values = np.asarray(values)
    norms = np.asarray(down_norms)
    coordinates = [tuple(u) for u in coordinates]
    positions = record['positions']
    if not positions or not coordinates or coordinates != sorted(set(coordinates)):
        raise ValueError('Nonempty aligned positions and unique sorted units required')
    if values.shape != (len(positions), len(coordinates)) or values.size > 2000000:
        raise ValueError('Selected values do not align with tokens and units')
    if norms.shape != (len(coordinates),) or not np.isfinite(values).all() or not np.isfinite(norms).all() or (norms < 0).any():
        raise ValueError('Finite measurements and nonnegative column norms required')
    if not isinstance(provenance, str) or not provenance.strip():
        raise ValueError('Explicit evidence provenance required')
    label = token_label or (lambda token: 'token ' + str(token))
    tokens = []
    for position in positions:
        processed, predicted = position['processed_token_id'], position['predicted_token_id']
        if type(processed) is not int or processed < 0 or predicted is not None and (type(predicted) is not int or predicted < 0):
            raise ValueError('Invalid token identity')

            raise ValueError('A measured decision requires its predicted token')
        processed_label = label(processed)
        predicted_label = label(predicted) if predicted is not None else 'no next-token measurement'
        if not isinstance(processed_label, str) or not isinstance(predicted_label, str):
            raise ValueError('Token labels must be text')
        tokens.append({**position, 'processed': processed_label, 'predicted': predicted_label})
    units = []
    for i, (layer, channel) in enumerate(coordinates):
        if type(layer) is not int or type(channel) is not int or min(layer, channel) < 0:
            raise ValueError('Invalid native coordinate')
        units.append({'layer': layer, 'channel': channel, 'down_norm': float(norms[i]), 'values': values[:, i].astype(float).tolist()})
    return {'schema': 'digital-surgery-trace-v1', 'provenance': provenance, 'private': True, 'tokens': tokens, 'units': units, 'scope': 'Selected actual channel values. def main():

"""Local gated-MLP edits and reversible channel steering.

Dense structural edits return a new module and leave the source untouched.
Packed 4-bit weights are explicitly unsupported here. Quantized model work must
dequantize the selected layer deliberately, or use a documented sparse branch.
Exact-arithmetic invariants do not imply bitwise floating-point equivalence.

from contextlib import contextmanager
import torch
from torch import nn


class DenseGatedMLP(nn.Module):
    def __init__(self, gate, up, down, activation):
        super().__init__()
        width, hidden = gate.shape
        if up.shape != gate.shape or down.shape != (hidden, width):
            raise ValueError('Aligned gate/up rows and down columns required')
        self.hidden_size, self.intermediate_size = hidden, width
        self.gate_proj = nn.Linear(hidden, width, bias=False, device=gate.device, dtype=gate.dtype)
        self.up_proj = nn.Linear(hidden, width, bias=False, device=gate.device, dtype=gate.dtype)
        self.down_proj = nn.Linear(width, hidden, bias=False, device=gate.device, dtype=gate.dtype)
        with torch.no_grad():
            self.gate_proj.weight.copy_(gate)
            self.up_proj.weight.copy_(up)
            self.down_proj.weight.copy_(down)
        self.act_fn = activation
        self.requires_grad_(False)

    def forward(self, x):
        return self.down_proj(self.act_fn(self.gate_proj(x)) * self.up_proj(x))


def _weights(mlp):
    projections = [getattr(mlp, name, None) for name in ['gate_proj', 'up_proj', 'down_proj']]
    if any(not isinstance(p, nn.Linear) or p.bias is not None for p in projections):
        raise ValueError('Plain bias-free dense projections required; unpack quantized/adapter weights explicitly')
    weights = [p.weight.detach() for p in projections]
    if any(not w.is_floating_point() or not torch.isfinite(w).all() for w in weights):
        raise ValueError('Finite unpacked floating-point matrices required')
    gate, up, down = weights
    if gate.ndim != 2 or up.shape != gate.shape or down.shape != (gate.shape[1], gate.shape[0]):
        raise ValueError('Inconsistent gated-MLP shapes')
    if len({(w.dtype, w.device) for w in weights}) != 1:
        raise ValueError('Shared dtype and device required')
    if not callable(getattr(mlp, 'act_fn', None)):
        raise ValueError('Explicit gated activation required')
    return gate, up, down


def _indices(values, width, duplicates=False):
    values = list(values)
    if not values or any(isinstance(x, bool) or not isinstance(x, int) or not 0 <= x < width for x in values):
        raise ValueError('Nonempty valid integer channel indices required')
    if not duplicates and len(set(values)) != len(values):
        raise ValueError('Duplicate channels are ambiguous for this operation')
    return values


def delete_channels(mlp, channels):
    """Physically remove linked gate/up rows and down columns in a dense copy."""
    gate, up, down = _weights(mlp)
    deleted = set(_indices(channels, gate.shape[0]))
    keep = [i for i in range(gate.shape[0]) if i not in deleted]
    if not keep:
        raise ValueError('Retain at least one channel')
    return DenseGatedMLP(gate[keep], up[keep], down[:, keep], mlp.act_fn)


def duplicate_channels(mlp, channels, preserve_function=True):
    """Copy input vectors; optionally split outgoing weights among all replicas.

    preserve_function=False is an amplification control, not new knowledge.
    """
    gate, up, down = _weights(mlp)
    extra = _indices(channels, gate.shape[0], duplicates=True)
    mapping = torch.tensor(list(range(gate.shape[0])) + extra, device=gate.device)
    new_down = down[:, mapping].clone()
    if preserve_function:
        multiplicity = torch.bincount(mapping, minlength=gate.shape[0])
        new_down /= multiplicity[mapping].to(dtype=down.dtype).unsqueeze(0)
    return DenseGatedMLP(gate[mapping], up[mapping], new_down, mlp.act_fn)


def add_channels(mlp, gate_rows, up_rows, down_columns):
    """Add genuinely specified new input/output vectors to a dense layer copy."""
    gate, up, down = _weights(mlp)
    if (gate_rows.ndim != 2 or gate_rows.shape[0] == 0 or gate_rows.shape[1] != gate.shape[1]
            or up_rows.shape != gate_rows.shape or down_columns.shape != (gate.shape[1], gate_rows.shape[0])):
        raise ValueError('New gate/up rows and output columns must have matching dimensions')
    added = [gate_rows, up_rows, down_columns]
    if any(not w.is_floating_point() or not torch.isfinite(w).all() for w in added):
        raise ValueError('Finite new floating-point vectors required')
    gate_rows, up_rows, down_columns = [w.to(device=gate.device, dtype=gate.dtype) for w in added]
    return DenseGatedMLP(torch.cat([gate, gate_rows]), torch.cat([up, up_rows]),
                         torch.cat([down, down_columns], dim=1), mlp.act_fn)


def edit_output_vectors(mlp, channels, deltas):
    """Alter only selected columns that write channel contributions to the stream."""
    gate, up, down = _weights(mlp)
    channels = _indices(channels, gate.shape[0])
    if deltas.shape != (down.shape[0], len(channels)) or not deltas.is_floating_point() or not torch.isfinite(deltas).all():

    changed = down.clone()
    changed[:, channels] += deltas.to(device=down.device, dtype=down.dtype)

class NewGatedNeurons(nn.Module):

class CopiedMemoryProjection(nn.Module):
    def __init__(self, source, target_index):

            if module is not None and (not isinstance(module, nn.Linear) or not module.weight.is_floating_point()
                                       or getattr(module.weight, 'quant_state', None) is not None):

        self.v_proj = copy.deepcopy(source.v_proj)

        keys = self.k_proj(hidden).view(shape)
        values = self.v_proj(hidden).view(shape) if self.v_proj is not None else keys
        keys = self.k_norm(keys)
        keys = apply_rotary_pos_emb(keys, *position_embeddings, unsqueeze_dim=2).transpose(1, 2)

                raise ValueError('Start a fresh sequence with the expanded cache allocation')

def _exact(value):

def _fraction(value):

def _mean(values):

def _manifest(rows):
    output = {}
    for row in rows:
        if not isinstance(row, dict) or set(row) != {'id', 'role', 'family'}:
            raise ValueError('Only explicit case identity, role and family required')
        if any(not isinstance(row[k], str) or not row[k].strip() for k in row):

def _records(rows, cases):
    output = {}
    for row in rows:
        if not isinstance(row, dict) or set(row) != FIELDS or not isinstance(row['id'], str) or row['id'] not in cases or row['id'] in output:
            raise ValueError('Exactly one complete record per immutable case ID required')
        if type(row['correct']) is not bool or type(row['at_cap']) is not bool:
            raise ValueError('Actual Boolean correctness and cap labels required')
        count = row['generated_tokens']
        if type(count) is not int or count <= 0:
            raise ValueError('A positive generated-token count required')
        for key in ['comment_tokens', 'unknown_comment_tokens']:

                raise ValueError('Nonnegative integer classified and unknown counts required')
        if row['comment_tokens'] + row['unknown_comment_tokens'] > count:
            raise ValueError('Classified and uncertain comment counts must be disjoint and bounded')
        if row['correct'] and row['at_cap']:
            raise ValueError('An unfinished capped response cannot be labelled correct')
        output[row['id']] = dict(row)

def _paired(ids, baseline, treatment):
    ratios, lower, upper = [], [], []
    gains, losses = [], []
    for identity in ids:
        b, t = baseline[identity], treatment[identity]
        ratios.append(Fraction(t['generated_tokens'], b['generated_tokens']))
        lower.append(Fraction(b['comment_tokens'], b['generated_tokens']) - Fraction(t['comment_tokens'] + t['unknown_comment_tokens'], t['generated_tokens']))
        upper.append(Fraction(b['comment_tokens'] + b['unknown_comment_tokens'], b['generated_tokens']) - Fraction(t['comment_tokens'], t['generated_tokens']))
        if t['correct'] and not b['correct']: gains.append(identity)
        if b['correct'] and not t['correct']: losses.append(identity)
    ratio, low, high = _mean(ratios), _mean(lower), _mean(upper)
    return {'cases': len(ids), 'original_correct': sum(baseline[i]['correct'] for i in ids),
            'correct': sum(treatment[i]['correct'] for i in ids), 'gains': gains, 'losses': losses,
            'caps': sum(treatment[i]['at_cap'] for i in ids),
            'generated_tokens': sum(treatment[i]['generated_tokens'] for i in ids),

    base = _records(baseline, cases)
    if not isinstance(conditions, dict) or not conditions or any(not isinstance(name, str) or not name.strip() for name in conditions):
        raise ValueError('Explicit named conditions required')

    retention = [i for i, c in cases.items() if c['role'] == 'retention']

def digest(text):

def observe(record, function_name):
    """Parse one already audited reply. Outer stripping/fences match extraction.

    Names, literals, docstrings, contexts and operators remain in the AST.
    Only line/column attributes are excluded from its structural comparison.
    This does not establish semantic equivalence or revalidate correctness.
    """
    if not isinstance(function_name, str) or not function_name.isidentifier():
        raise ValueError('Explicit valid function name required')
    if record.get('complete') is not True or record.get('termination') != 'eos':
        raise ValueError('Only completed native-EOS replies may be described')
    tokens = record.get('generated_ids')
    if (not isinstance(tokens, list) or not tokens
            or any(type(token) is not int or token < 0 for token in tokens)):
        raise ValueError('Nonempty integer generated-token sequence required')
    if type(record.get('generated_tokens')) is not int or record['generated_tokens'] != len(tokens):
        raise ValueError('Saved token count mismatch')
    text = record.get('text')
    if not isinstance(text, str) or len(text) > 200_000:
        raise ValueError('Bounded saved text required')
    source = text.strip()
    if source.startswith('```'):
        match = re.fullmatch(r'```(?:python|python3|py)?\s*\n([\s\S]*?)\n```', source)
        if not match:
            raise ValueError('One complete Python/plain fence without outside prose required')
        source = match[1]
    tree = ast.parse(source)
    functions = [node for node in tree.body if isinstance(node, ast.FunctionDef)]
    if len(functions) != 1 or functions[0].name != function_name:
        raise ValueError('Exactly one complete expected function required')
    if any(not isinstance(node, (ast.Import, ast.ImportFrom, ast.FunctionDef))
           and not (isinstance(node, ast.Expr) and isinstance(node.value, ast.Constant)
                    and isinstance(node.value.value, str)) for node in tree.body):
        raise ValueError('Outside prose or nondefinition module statement refused')
    normalized = ast.dump(tree, annotate_fields=True, include_attributes=False)
    operators = Counter()
    controls = Counter()
    for node in ast.walk(tree):
        if isinstance(node, (ast.operator, ast.unaryop, ast.boolop, ast.cmpop)):
            operators[type(node).__name__] += 1
        if type(node).__name__ in CONTROL_NODES:
            controls[type(node).__name__] += 1
    return {
        'tokens': tuple(tokens), 'text': text, 'source': source, 'AST': normalized,
        'token_sha256': digest(json.dumps(tokens, separators=(',', ':'))),
        'source_sha256': digest(source), 'AST_sha256': digest(normalized),
        'operator_nodes': dict(sorted(operators.items())),
        'control_nodes': {key: controls[key] for key in CONTROL_NODES},

def saved_outcome(original, changed):
    """Label existing grades; never execute a candidate or call an oracle."""
    for grade in (original, changed):
        if type(grade.get('correct')) is not bool or grade.get('source_valid') is not True:
            raise ValueError('Exact existing source-valid boolean grades required')
        if grade.get('status') != ('passed' if grade['correct'] else 'wrong_answer'):
            raise ValueError('Only complete saved executable coding outcomes accepted')
    return {
        (True, True): 'preserved_correct', (False, False): 'both_wrong',
        (False, True): 'gain', (True, False): 'loss',
    }[original['correct'], changed['correct']]


def compare(original, changed, original_grade, changed_grade):
    """Partition one pair in the fixed protocol order, preserving all outcomes."""
    tokens_same = original['tokens'] == changed['tokens']
    source_same = original['source'] == changed['source']
    ast_same = original['AST'] == changed['AST']
    if tokens_same and original['text'] != changed['text']:
        raise ValueError('Identical token IDs with conflicting decoded text')
    if tokens_same:
        category = CATEGORIES[0]
    elif source_same:
        category = CATEGORIES[1]
    elif ast_same:
        category = CATEGORIES[2]
    else:
        category = CATEGORIES[3]
    if (source_same and not ast_same) or (tokens_same and not source_same):
        raise ValueError('Contradictory observation lineage')
    outcome = saved_outcome(original_grade, changed_grade)
    if ast_same and outcome in ('gain', 'loss'):
        raise ValueError('Identical AST with conflicting saved deterministic grades')
    return {
        'category': category, 'saved_outcome': outcome,

"""Greedy native-EOS generation with streamed evidence and resumable history.

There is no response-token ceiling. The caller supplies actual resource checks;
the model's configured context is a separate boundary, not a completion signal.
An interrupted record remains incomplete. Pretrained use requires a prospective
protocol binding the prompt, model, edit and any saved native reference history.
"""

import copy

import torch
from transformers.cache_utils import DynamicCache

from research.experiments.n37.continuation import decision

RESOURCE_REASONS = {'wall_time_interrupt', 'memory_reservation_interrupt',
                    'output_storage_interrupt', 'external_interrupt'}


def generate_native_eos(model, tokenizer, encoded, eos, *, resource_guard,
                        on_decision, reference=None):
    """Continue to native EOS, recording actual resource/context interruptions.

    ``resource_guard`` returns None or a JSON-compatible resource witness with
    ``reason`` and ``evidence``. It must measure resources, not impose a disguised
    token limit. ``on_decision`` persists every new token and its exact hashes.
    A reference is unfinished own-history evidence, never a donor or an answer.
    Every reference choice/logit/cache must replay exactly before a new suffix.
    """
    config = model.config.text_config
    maximum = config.max_position_embeddings
    if type(maximum) is not int or maximum < 1:
        raise ValueError('Actual configured context boundary required')
    if not callable(resource_guard) or not callable(on_decision):
        raise ValueError('Resource and streaming callbacks required')
    if not eos or any(type(i) is not int or not 0 <= i < config.vocab_size for i in eos):
        raise ValueError('Explicit valid native EOS IDs required')
    if any(p.requires_grad or p.grad is not None for p in model.parameters()):
        raise ValueError('Freeze model parameters before inference')
    if any(module.training for module in model.modules()):
        raise ValueError('Evaluation mode required')
    versions = {n: p._version for n, p in model.named_parameters()}
    n = int(encoded['input_ids'].numel())
    if encoded['input_ids'].ndim != 2 or encoded['input_ids'].shape[0] != 1 or n < 1:
        raise ValueError('One nonempty encoded native prompt required')
    prefix = [] if reference is None else list(reference['generated_ids'])
    old_trace = [] if reference is None else copy.deepcopy(reference['decision_trace'])
    if reference is not None:
        if not prefix or len(prefix) != len(old_trace) or any(i in eos for i in prefix):
            raise ValueError('Only unfinished non-EOS own histories can resume')
        if prefix != [r['argmax'] for r in old_trace]:
            raise ValueError('Reference token choices and trace disagree')
    ids, trace, replay = list(prefix), list(old_trace), []
    first = None if reference is None else copy.deepcopy(reference['first_cache'])
    layers = config.num_hidden_layers - config.num_kv_shared_layers
    dtype = str(next(model.parameters()).dtype)
    cache = DynamicCache(config=config)
    attempted = exposures = completed = emitted = 0
    step = 0
    interruption = None
    termination = 'interrupted'
    try:
        with torch.no_grad():
            while True:
                interruption = resource_guard()
                if interruption is not None:
                    if not isinstance(interruption, dict) or interruption.get('reason') not in RESOURCE_REASONS or 'evidence' not in interruption:
                        raise ValueError('Explicit actual-resource interruption witness required')
                    break
                if n + step > maximum:
                    interruption = {'reason': 'context_exhausted',
                                    'evidence': {'configured_context': maximum,
                                                 'next_required_input_length': n + step}}
                    break
                args = encoded if step == 0 else {'input_ids': torch.tensor(
                    [[prefix[step - 1] if step < len(prefix) else ids[-1]]],
                    device=encoded['input_ids'].device)}
                attempted += 1
                exposures += n if step == 0 else 1
                try:
                    output = model(**args, past_key_values=cache, use_cache=True, logits_to_keep=1)
                    item, witness = decision(output, step, layers, dtype)
                except torch.OutOfMemoryError as error:
                    interruption = {'reason': 'memory_exhausted', 'evidence': {
                        'exception_type': type(error).__name__, 'during_forward_or_witness': True}}
                    break
                completed += 1
                cache = output.past_key_values
                if step == 0:
                    if reference is not None:
                        assert witness == reference['first_cache'], 'Native reference first cache changed'
                    first = witness
                if step < len(prefix):
                    assert item == old_trace[step], 'Native own-history choice/logit/cache changed'
                    replay.append(item)
                else:
                    assert len(replay) == len(prefix)
                    ids.append(item['argmax'])
                    trace.append(item)
                    emitted += 1
                    on_decision(copy.deepcopy(item))
                    if item['argmax'] in eos:
                        termination = 'eos'
                        interruption = None
                        break
                step += 1
    finally:
        assert versions == {n: p._version for n, p in model.named_parameters()}
        assert all(not p.requires_grad and p.grad is None for p in model.parameters())
    return {'generated_ids': ids, 'text': tokenizer.decode(ids, skip_special_tokens=True),
            'generated_tokens': len(ids), 'new_generated_tokens': emitted,
            'decision_trace': trace, 'first_cache': first, 'replayed_trace': replay,
            'reference_tokens_replayed_exact': len(replay), 'reference_tokens_required': len(prefix),
            'prefix_exact': len(replay) == len(prefix), 'termination': termination,
            'complete': termination == 'eos', 'interruption': interruption,
            'input_tokens': n, 'attempted_forward_calls': attempted,
            'completed_forward_calls': completed, 'input_token_exposures': exposures,
            'new_response_token_ceiling': None, 'configured_context': maximum,
            'optimizer_steps': 0, 'backward_calls': 0, 'parameter_assignments': 0}
