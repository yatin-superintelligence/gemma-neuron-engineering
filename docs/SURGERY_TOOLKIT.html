"""Clone an actual native past for one current-token causal measurement.

The clone is ephemeral. It is not a reusable past for a changed conversation.
Only the pinned DynamicCache with initialized dense KV layers is accepted.
"""

import copy
import hashlib
import json
from contextlib import contextmanager

import torch
from transformers.cache_utils import DynamicCache, DynamicLayer, DynamicSlidingWindowLayer


def tensor_hash(value):
    return hashlib.sha256(value.detach().contiguous().reshape(-1).view(torch.uint8).cpu().numpy().tobytes()).hexdigest()


def _value(value):
    if torch.is_tensor(value):
        return {'shape': list(value.shape), 'dtype': str(value.dtype), 'device': str(value.device), 'sha256': tensor_hash(value)}
    if value is None or type(value) in (str, bool, int, float):
        return value
    if isinstance(value, (torch.dtype, torch.device)):
        return str(value)
    raise ValueError('Unsupported mutable cache metadata')


def fingerprint(cache, entries, length, dtype):
    """Validate logical and physical layout and retain exact compact witnesses."""
    if type(cache) is not DynamicCache or cache.offloading or cache.layer_class_to_replicate is not None:
        raise ValueError('Initialized non-offloaded DynamicCache required')
    if type(entries) is not int or type(length) is not int or min(entries, length) < 1 or len(cache.layers) != entries:
        raise ValueError('Exact initialized cache cardinality and length required')
    if set(cache.__dict__) != {'layers', 'layer_class_to_replicate', 'offloading'}:
        raise ValueError('Unexpected cache implementation state')
    rows = []
    for index, layer in enumerate(cache.layers):
        if type(layer) not in (DynamicLayer, DynamicSlidingWindowLayer) or not layer.is_initialized:
            raise ValueError('Actual initialized native layer required')
        if int(layer.get_seq_length()) != length:
            raise ValueError('Logical cache length differs from declared native past')
        k, v = layer.keys, layer.values
        if not torch.is_tensor(k) or not torch.is_tensor(v) or k.shape != v.shape or k.ndim != 4 or k.shape[0] != 1:
            raise ValueError('Dense batch-one aligned key/value tensors required')
        physical = min(length, layer.sliding_window - 1) if type(layer) is DynamicSlidingWindowLayer else length
        if k.shape[-2] != physical or min(k.shape) < 1 or k.dtype != dtype or v.dtype != dtype or k.device != v.device:
            raise ValueError('Truncated, mismatched or wrong-dtype native cache')
        if not torch.isfinite(k).all() or not torch.isfinite(v).all():
            raise ValueError('Nonfinite native cache')
        rows.append({'layer': index, 'type': type(layer).__name__, 'attributes': {key: _value(value) for key, value in sorted(layer.__dict__.items())}})
    raw = json.dumps(rows, sort_keys=True, separators=(',', ':'))
    return {'entries': entries, 'logical_length': length, 'layers': rows, 'sha256': hashlib.sha256(raw.encode()).hexdigest()}


def _storages(cache):
    return {value.untyped_storage().data_ptr() for layer in cache.layers for value in layer.__dict__.values() if torch.is_tensor(value) and value.numel()}


@contextmanager
def native_cache_branch(cache, entries, length, dtype):
    """Yield a fresh nonaliasing native past and require one-token consumption.

    The source must stay byte-exact, including sliding-window counters. A failed
    probe still checks source immutability. Full raw pasts are never serialized.
    """
    before = fingerprint(cache, entries, length, dtype)
    branch = copy.deepcopy(cache)
    if fingerprint(branch, entries, length, dtype) != before or _storages(cache) & _storages(branch):
        raise ValueError('Cache copy is inexact or aliases native storage')
    success = False
    try:
        yield branch
        success = True
    finally:
        if fingerprint(cache, entries, length, dtype) != before:
            raise AssertionError('Native source past changed during measurement')
        if success:
            fingerprint(branch, entries, length + 1, dtype)

"""Read native MLP activity without changing its forward tensors or parameters."""

import torch


def spaced_positions(length, maximum):
    """Choose deterministic integer endpoints without response-dependent selection."""
    if not isinstance(length, int) or not isinstance(maximum, int):
        raise ValueError('Integer lengths required')
    if min(length, maximum) < 1:
        raise ValueError('Positive lengths required')
    count = min(length, maximum)
    return [0] if count == 1 else [i * (length - 1) // (count - 1) for i in range(count)]


class NativeActivity:
    """Measure post-gating channels and both sides of post-MLP normalization.

    This observer retains selected token slices and streaming moments. It never
    returns a replacement forward value. Call ``begin`` before each model call
    and ``finish`` afterward. Moments cover only the last processed position of
    each call, avoiding extra weight for long prompt prefills.
    """

    def __init__(self, model, coordinates):
        self.layers = list(model.model.language_model.layers)
        self.coordinates = [tuple(x) for x in coordinates]
        if not self.coordinates or self.coordinates != sorted(set(self.coordinates)):
            raise ValueError('Coordinates must be nonempty, unique and sorted')
        self.widths = [layer.mlp.down_proj.in_features for layer in self.layers]
        for layer, channel in self.coordinates:
            if type(layer) is not int or type(channel) is not int:
                raise ValueError('Integer coordinates required')
            if not 0 <= layer < len(self.layers) or not 0 <= channel < self.widths[layer]:
                raise ValueError('Coordinate outside native MLP')
        self.indices = {
            layer: [channel for li, channel in self.coordinates if li == layer]
            for layer in range(len(self.layers))
        }
        with torch.no_grad():
            self.down_norms = [layer.mlp.down_proj.weight.detach().float().norm(dim=0) for layer in self.layers]
        if not all(torch.isfinite(x).all() for x in self.down_norms):
            raise ValueError('Nonfinite native output columns')
        self.handles = []
        self.active = None
        self.reset_moments()

    def reset_moments(self):
        if self.active is not None:
            raise ValueError('Cannot reset an unfinished forward')
        self.steps = 0
        self.sums = [torch.zeros((3, w), dtype=torch.float64, device=n.device) for w, n in zip(self.widths, self.down_norms)]

    def begin(self, sequence_length, positions, toplist=False):
        if not self.handles or self.active is not None:
            raise ValueError('Observer is closed or a forward is unfinished')
        if type(sequence_length) is not int or sequence_length < 1:
            raise ValueError('Positive native sequence length required')
        positions = list(positions)
        if positions != sorted(set(positions)) or any(type(i) is not int or not 0 <= i < sequence_length for i in positions):
            raise ValueError('Invalid sampled positions')
        self.active = {'length': sequence_length, 'positions': positions, 'toplist': bool(toplist), 'h': {}, 'raw': {}, 'norm': {}, 'tops': {}}

    def _check(self, value, layer, key, width):
        frame = self.active
        if frame is None or layer in frame[key]:
            raise ValueError('Missing begin or repeated native module call')
        if value.ndim != 3 or tuple(value.shape) != (1, frame['length'], width):
            raise ValueError('Unexpected native tensor shape')
        if not torch.isfinite(value).all():
            raise ValueError('Nonfinite observed tensor')
        return frame

    @torch.no_grad()
    def _input(self, layer, module, args):
        value = args[0].detach()
        frame = self._check(value, layer, 'h', self.widths[layer])
        current = value[0, -1].double()
        self.sums[layer][0].add_(current)
        self.sums[layer][1].add_(current.abs())
        self.sums[layer][2].add_(current.square())
        sample = value[0, frame['positions']].float()
        picked = sample[:, self.indices[layer]]
        stats = torch.stack((sample.mean(-1), sample.square().mean(-1).sqrt(), sample.abs().amax(-1)), dim=-1)
        frame['h'][layer] = (picked.cpu(), stats.cpu())
        if frame['toplist']:
            magnitudes = current.float().abs() * self.down_norms[layer]
            order = torch.argsort(magnitudes, descending=True, stable=True)[:min(8, self.widths[layer])]
            frame['tops'][layer] = {
                'channels': order.cpu().tolist(),
                'activation': current[order].float().cpu().tolist(),
                'raw_projection_magnitude': magnitudes[order].cpu().tolist(),
            }

    @torch.no_grad()
    def _output(self, layer, key, module, args, output):
        value = output.detach()
        width = self.layers[layer].mlp.down_proj.out_features
        frame = self._check(value, layer, key, width)
        frame[key][layer] = value[0, frame['positions']].float().norm(dim=-1).cpu()

    def finish(self):
        frame = self.active
        if frame is None:
            raise ValueError('No forward is active')
        expected = set(range(len(self.layers)))
        if any(set(frame[key]) != expected for key in ['h', 'raw', 'norm']):
            raise ValueError('Incomplete native module coverage')
        if frame['toplist'] and set(frame['tops']) != expected:
            raise ValueError('Incomplete native toplist coverage')
        units = torch.cat([frame['h'][i][0] for i in range(len(self.layers))], dim=1)
        stats = torch.stack([
            torch.cat((frame['h'][i][1], frame['raw'][i][:, None], frame['norm'][i][:, None]), dim=-1)
            for i in range(len(self.layers))
        ], dim=1)
        self.active = None
        self.steps += 1
        return {'values': units.numpy(), 'layer_stats': stats.numpy(), 'tops': frame['tops']}

    def moments(self):
        if self.active is not None or self.steps < 1:
            raise ValueError('No complete trajectory moments')
        return torch.cat(self.sums, dim=1) / self.steps

    def __enter__(self):
        if self.handles:
            raise ValueError('Observer already installed')
        try:
            for i, layer in enumerate(self.layers):
                def pre(module, args, i=i):
                    self._input(i, module, args)
                def raw(module, args, output, i=i):
                    self._output(i, 'raw', module, args, output)
                def norm(module, args, output, i=i):
                    self._output(i, 'norm', module, args, output)
                self.handles.append(layer.mlp.down_proj.register_forward_pre_hook(pre))
                self.handles.append(layer.mlp.down_proj.register_forward_hook(raw))
                self.handles.append(layer.post_feedforward_layernorm.register_forward_hook(norm))
        except BaseException:
            self.__exit__(None, None, None)
            raise
        return self

    def __exit__(self, exc_type, exc, tb):
        unfinished = self.active is not None
        for handle in self.handles:
            handle.remove()
        self.handles = []
        self.active = None
        if unfinished and exc_type is None:
            raise ValueError('Observer exited before a forward finished')

"""PyTorch MLP-channel measurement and reversible interventions.

Use ordinary eager Transformers/PEFT execution. Some optimized kernels bypass
module hooks; finish() fails if any expected hook was not observed.
"""
from contextlib import contextmanager
import re

import torch


def mlp_modules(model):
    candidates={name:module for name,module in model.named_modules()
                if name.endswith('.mlp.down_proj')}
    # Multimodal Gemma also has vision MLPs with the same projection names.
    # Prefer the explicitly named text decoder; text-only models need no prefix.
    text={name:module for name,module in candidates.items()
          if re.search(r'(^|\.)language_model\.layers\.\d+\.mlp\.down_proj$',name)}
    return text if text else candidates


class ActivationRecorder:
    def __init__(self,model):
        self.modules=mlp_modules(model)
        if not self.modules:raise ValueError('No MLP down projections found')
        self.handles=[];self.stats={};self.mask=None

    def set_mask(self,attention_mask):
        self.mask=attention_mask

    def _hook(self,name):
        def capture(module,args):
            values=args[0].detach().float()
            if values.ndim!=3:raise ValueError('Require [batch, tokens, channels] eager activations')
            if self.mask is None or tuple(self.mask.shape)!=tuple(values.shape[:2]):
                raise ValueError('Provide a matching token mask; free-generation traces are not aligned probes')
            selected=values[self.mask.to(device=values.device,dtype=torch.bool)]
            if not len(selected):raise ValueError('Probe contains no unmasked tokens')
            count=selected.shape[0]
            if name not in self.stats:
                self.stats[name]={'count':0,'sum':torch.zeros(values.shape[-1],device=values.device),
                    'sum_abs':torch.zeros(values.shape[-1],device=values.device),
                    'sum_sq':torch.zeros(values.shape[-1],device=values.device)}
            item=self.stats[name];item['count']+=count
            item['sum']+=selected.sum(dim=0)
            item['sum_abs']+=selected.abs().sum(dim=0)
            item['sum_sq']+=(selected*selected).sum(dim=0)
        return capture

    def __enter__(self):
        self.handles=[module.register_forward_pre_hook(self._hook(name)) for name,module in self.modules.items()]
        return self

    def finish(self):
        missing=set(self.modules)-set(self.stats)
        if missing:raise RuntimeError('Unobserved hooks; optimized execution may bypass instrumentation: '+str(sorted(missing)))
        result={}
        for name,item in self.stats.items():
            if any(not torch.isfinite(item[key]).all().item() for key in ('sum','sum_abs','sum_sq')):
                raise RuntimeError('Nonfinite activation statistics at '+name)
            n=item['count'];mean=item['sum']/n
            result[name]={'token_count':n,'mean':mean.cpu(),
                'mean_abs':(item['sum_abs']/n).cpu(),
                'rms':(item['sum_sq']/n).sqrt().cpu(),
                'variance':(item['sum_sq']/n-mean*mean).clamp_min(0).cpu()}
        return result

    def __exit__(self,*args):
        for handle in self.handles:handle.remove()
        self.handles=[]

"""Describe output text regions without grading or executing a program."""

import ast
import re


def python_regions(text):
    pattern = re.compile(r'```(?:python3?|py)?[ \t]*\r?\n(?P<body>.*?)```', re.I | re.S)
    regions = [list(m.span('body')) for m in pattern.finditer(text)]
    if not regions:
        try:
            tree = ast.parse(text)
        except (SyntaxError, ValueError):
            tree = None
        if tree and any(isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef)) for n in ast.walk(tree)):
            regions = [[0, len(text)]]
    return regions


def label_prefixes(text, prefixes, token_ids, special_ids, coding=True):
    """Unstable Unicode prefix decoding remains explicitly unaligned.

    Labels describe textual regions, not executed computation or neuron roles.
    The caller supplies decoded native prefixes; no generation occurs here.
    """
    if len(prefixes) != len(token_ids) or not prefixes or prefixes[-1] != text:
        raise ValueError('Complete exact native decoding is required')
    regions = python_regions(text) if coding else []
    previous, previous_ok = '', True
    result = []
    for index, (decoded, token) in enumerate(zip(prefixes, token_ids)):
        current_ok = text.startswith(decoded)
        start, end = len(previous), len(decoded)
        aligned = previous_ok and current_ok and start <= end
        if token in special_ids:
            label = 'special'
        elif not aligned:
            label = 'unaligned'
        elif start == end:
            label = 'zero_width'
        elif not coding:
            label = 'retention_content'
        elif any(a <= start and end <= b for a, b in regions):
            label = 'python_region'
        elif any(start < b and end > a for a, b in regions):
            label = 'boundary'
        else:

def lexical_regions(text, regions):
    output, errors = [], []
    for start, end in regions:
        body = text[start:end]
        lines = body.splitlines(keepends=True)
        offsets = [0]
        for line in lines:
            offsets.append(offsets[-1] + len(line))
        def absolute(position):
            line, column = position
            return start + offsets[min(line - 1, len(offsets) - 1)] + column
        try:
            for item in tokenize.generate_tokens(io.StringIO(body).readline):
                kind = {
                    token.COMMENT: 'comment', token.STRING: 'string',
                    token.NUMBER: 'number', token.OP: 'operator',
                    token.NAME: 'keyword' if keyword.iskeyword(item.string) else 'name',
                }.get(item.type)
                if kind is not None:

                    output.append({'start': a, 'end': b, 'category': kind})
        except (tokenize.TokenError, IndentationError, SyntaxError) as error:
            errors.append({'region_start': start, 'type': type(error).__name__})
    return output, errors

def compose(norm, raw, reciprocal):
    """Use recorded native factors, the unchanged gamma and native operation order."""
    check_norm(norm)
    if raw.dtype not in (torch.float32, torch.bfloat16) or raw.requires_grad:
        raise ValueError('Detached FP32/BF16 raw native state required')
    if raw.ndim != 3 or raw.shape[:2] != (1, 1) or raw.shape[-1] != norm.weight.numel():
        raise ValueError('One explicit current-position branch required')
    if reciprocal.dtype != torch.float32 or reciprocal.shape != (1, 1, 1) or reciprocal.requires_grad:
        raise ValueError('Detached FP32 native reciprocal RMS required')
    if raw.device != norm.weight.device or reciprocal.device != raw.device:
        raise ValueError('Native factor devices must agree')
    if (not torch.isfinite(raw).all() or not torch.isfinite(reciprocal).all()
            or not torch.all(reciprocal > 0)):
        raise ValueError('Finite raw state and positive native reciprocal RMS required')
    with torch.no_grad():
        return ((raw.float() * reciprocal) * norm.weight.float()).type_as(raw)

Dense structural edits return a new module and leave the source untouched.
Packed 4-bit weights are explicitly unsupported here. Quantized model work must
dequantize the selected layer deliberately, or use a documented sparse branch.
Exact-arithmetic invariants do not imply bitwise floating-point equivalence.

from contextlib import contextmanager
import torch
from torch import nn


class DenseGatedMLP(nn.Module):
    def __init__(self, gate, up, down, activation):
        super().__init__()
        width, hidden = gate.shape
        if up.shape != gate.shape or down.shape != (hidden, width):
            raise ValueError('Aligned gate/up rows and down columns required')
        self.hidden_size, self.intermediate_size = hidden, width
        self.gate_proj = nn.Linear(hidden, width, bias=False, device=gate.device, dtype=gate.dtype)
        self.up_proj = nn.Linear(hidden, width, bias=False, device=gate.device, dtype=gate.dtype)
        self.down_proj = nn.Linear(width, hidden, bias=False, device=gate.device, dtype=gate.dtype)
        with torch.no_grad():
            self.gate_proj.weight.copy_(gate)
            self.up_proj.weight.copy_(up)
            self.down_proj.weight.copy_(down)
        self.act_fn = activation
        self.requires_grad_(False)

    def forward(self, x):
        return self.down_proj(self.act_fn(self.gate_proj(x)) * self.up_proj(x))


def _weights(mlp):
    projections = [getattr(mlp, name, None) for name in ['gate_proj', 'up_proj', 'down_proj']]
    if any(not isinstance(p, nn.Linear) or p.bias is not None for p in projections):
        raise ValueError('Plain bias-free dense projections required; unpack quantized/adapter weights explicitly')
    weights = [p.weight.detach() for p in projections]
    if any(not w.is_floating_point() or not torch.isfinite(w).all() for w in weights):
        raise ValueError('Finite unpacked floating-point matrices required')
    gate, up, down = weights
    if gate.ndim != 2 or up.shape != gate.shape or down.shape != (gate.shape[1], gate.shape[0]):
        raise ValueError('Inconsistent gated-MLP shapes')
    if len({(w.dtype, w.device) for w in weights}) != 1:
        raise ValueError('Shared dtype and device required')
    if not callable(getattr(mlp, 'act_fn', None)):
        raise ValueError('Explicit gated activation required')
    return gate, up, down


def _indices(values, width, duplicates=False):
    values = list(values)
    if not values or any(isinstance(x, bool) or not isinstance(x, int) or not 0 <= x < width for x in values):
        raise ValueError('Nonempty valid integer channel indices required')
    if not duplicates and len(set(values)) != len(values):
        raise ValueError('Duplicate channels are ambiguous for this operation')
    return values


def delete_channels(mlp, channels):
    """Physically remove linked gate/up rows and down columns in a dense copy."""
    gate, up, down = _weights(mlp)
    deleted = set(_indices(channels, gate.shape[0]))
    keep = [i for i in range(gate.shape[0]) if i not in deleted]
    if not keep:
        raise ValueError('Retain at least one channel')
    return DenseGatedMLP(gate[keep], up[keep], down[:, keep], mlp.act_fn)


def duplicate_channels(mlp, channels, preserve_function=True):
    """Copy input vectors; optionally split outgoing weights among all replicas.

    preserve_function=False is an amplification control, not new knowledge.
    """
    gate, up, down = _weights(mlp)
    extra = _indices(channels, gate.shape[0], duplicates=True)
    mapping = torch.tensor(list(range(gate.shape[0])) + extra, device=gate.device)
    new_down = down[:, mapping].clone()
    if preserve_function:
        multiplicity = torch.bincount(mapping, minlength=gate.shape[0])
        new_down /= multiplicity[mapping].to(dtype=down.dtype).unsqueeze(0)
    return DenseGatedMLP(gate[mapping], up[mapping], new_down, mlp.act_fn)


def add_channels(mlp, gate_rows, up_rows, down_columns):

def tensor_hash(value):
    return hashlib.sha256(value.detach().contiguous().view(torch.uint8).cpu().numpy().tobytes()).hexdigest()


class CopiedMemoryProjection(nn.Module):
    def __init__(self, source, target_index):
        super().__init__()
        for name in ['k_proj', 'v_proj']:
            module = getattr(source, name, None)
            if module is not None and (not isinstance(module, nn.Linear) or not module.weight.is_floating_point()
                                       or getattr(module.weight, 'quant_state', None) is not None):
                raise ValueError('Unpacked floating-point source projections required')
        self.k_proj = copy.deepcopy(source.k_proj)
        self.v_proj = copy.deepcopy(source.v_proj)
        self.k_norm = copy.deepcopy(source.k_norm)
        self.v_norm = copy.deepcopy(source.v_norm)
        self.head_dim = source.head_dim
        self.target_index = target_index
        self.requires_grad_(False)

    def forward(self, hidden, position_embeddings, cache):
        shape = (*hidden.shape[:-1], -1, self.head_dim)
        keys = self.k_proj(hidden).view(shape)
        values = self.v_proj(hidden).view(shape) if self.v_proj is not None else keys
        keys = self.k_norm(keys)
        keys = apply_rotary_pos_emb(keys, *position_embeddings, unsqueeze_dim=2).transpose(1, 2)
        values = self.v_norm(values).transpose(1, 2)
        if cache is not None:
            if not hasattr(cache, 'layers') or len(cache.layers) <= self.target_index:
                raise ValueError('Start a fresh sequence with the expanded cache allocation')
            keys, values = cache.update(keys, values, self.target_index)
        return keys, values

def memory_connection(model, source_index=14, target_index=24, strength=.25, mode='value_history'):

    Each sequence must start with a fresh cache. Do not reuse a cache across
    different policies or after leaving this context. Only causal full-attention
    source/target pairs are supported by this prototype.

    modes = {'kv', 'key_only', 'value_history', 'value_current', 'value_permuted'}
    if mode not in modes or not math.isfinite(strength) or not 0 <= strength <= 1:
        raise ValueError('Choose a declared mode and a fixed strength between zero and one')
    decoder = model.model.language_model
    layers = decoder.layers
    if not 0 <= source_index < target_index < len(layers):
        raise ValueError('Ordered valid source and target layers required')

        source.layer_type != 'full_attention' or target.layer_type != 'full_attention' or
        source.head_dim != target.head_dim):
        raise ValueError('Require an original full-attention producer and a compatible shared reader')
    if any(p.requires_grad or p.grad is not None for p in model.parameters()):
        raise ValueError('The parent model must be frozen with no parameter gradients')
    name = '_neuron_memory_connection'
    if hasattr(model, name):
        raise ValueError('A memory connection is already installed')
    widths = [layer.mlp.down_proj.in_features for layer in layers]
    flags = [(layer.self_attn.is_kv_shared_layer, layer.self_attn.store_full_length_kv) for layer in layers]
    original = {key: parameter._version for key, parameter in model.named_parameters()}
    configs = {id(config): config for config in [model.config.text_config, decoder.config]}
    shared_counts = {key: config.num_kv_shared_layers for key, config in configs.items()}
    branch = CopiedMemoryProjection(source, target_index).eval()
    copies = {}
    for key, parameter in branch.named_parameters():
        expected = source.get_parameter(key)
        assert torch.equal(parameter, expected)
        copies[key] = tensor_hash(parameter)
    receipt = {'source':source_index,'target':target_index,'mode':mode,'strength':strength,'calls':0,
        'projection_calls':0,'copied_parameter_hashes':copies,

        'MLP_widths_changed':False,'fixed_input_path_scope':True}

        receipt['calls'] += 1
        if strength == 0:
            return None

        shared = kwargs.get('shared_kv_states')

        if not torch.isfinite(new_keys).all() or not torch.isfinite(new_values).all():
            raise ValueError('Nonfinite copied memory')
        receipt['projection_calls'] += 1
        def blend(old, new):
            return new if strength == 1 else torch.lerp(old, new, strength)
        keys, values = old_keys, old_values
        if mode in ['kv', 'key_only']:
            keys = blend(old_keys, new_keys)
        if mode in ['kv', 'value_history']:
            values = blend(old_values, new_values)
        elif mode == 'value_permuted':
            values = blend(old_values, new_values.roll(shifts=1, dims=-1))
        elif mode == 'value_current':
            values = old_values.clone()
            values[:, :, -1:] = blend(old_values[:, :, -1:], new_values[:, :, -1:])
        shared[target.layer_type] = (keys, values)
        return None
    handle = None
    cache_handle = None
    cache_config = PreTrainedConfig(num_hidden_layers=len(layers),
        layer_types=list(decoder.config.layer_types),sliding_window=decoder.config.sliding_window)
    receipt['cache_expansions'] = 0
    def provision_cache(module, arguments, kwargs):
        if not kwargs.get('use_cache', True):
            return None
        cache = kwargs.get('past_key_values')
        if cache is None:
            cache = DynamicCache(config=cache_config)
            cache.layers = cache.layers[:target_index+1]
            kwargs = dict(kwargs)
            kwargs['past_key_values'] = cache
            receipt['cache_expansions'] += 1
            return arguments, kwargs
        if not isinstance(cache, DynamicCache):
            raise ValueError('This prototype requires a DynamicCache')
        if len(cache.layers) <= target_index:
            if cache.get_seq_length() != 0:
                raise ValueError('Cannot add historical memory to an already populated native cache')
            fresh = DynamicCache(config=cache_config)
            cache.layers.extend(fresh.layers[len(cache.layers):target_index+1])
            receipt['cache_expansions'] += 1
        if strength and cache.get_seq_length() != cache.get_seq_length(target_index):
            raise ValueError('Start a fresh sequence for each connection policy')
        return None
    try:
        # Cache-only metadata avoids both a zero-count slicing corner case and
        # accidental reconstruction of the parent's double-wide MLP shapes.
        model.add_module(name, branch)
        cache_handle = model.register_forward_pre_hook(provision_cache,with_kwargs=True)

        assert widths == [layer.mlp.down_proj.in_features for layer in layers]
        yield receipt
    finally:
        if handle is not None:
            handle.remove()
        if cache_handle is not None:
            cache_handle.remove()
        if hasattr(model, name):
            delattr(model, name)
        for key, config in configs.items():
            config.num_kv_shared_layers = shared_counts[key]

        assert flags == [(layer.self_attn.is_kv_shared_layer, layer.self_attn.store_full_length_kv) for layer in layers]
        assert original == {key:p._version for key,p in model.named_parameters()}
        receipt['restored_configuration_and_original_parameter_versions'] = True

def file_sha256(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b''):
            digest.update(block)
    return digest.hexdigest()


def tensor_sha256(tensor):
    raw = tensor.detach().contiguous().view(torch.uint8).cpu().numpy().tobytes()
    return hashlib.sha256(raw).hexdigest()


def inspect_bundle(folder):
    folder = Path(folder)
    manifest = json.loads((folder / 'bundle.json').read_text())
    if manifest['schema_version'] != 1 or manifest['optimizer_steps'] != 0:
        raise ValueError('Unsupported bundle or nonzero training steps')
    for name, digest in manifest['files'].items():

            raise ValueError('Bundle file/hash mismatch: ' + name)

def dense_target(mlp):
    weights = []
    for name in ['gate_proj', 'up_proj', 'down_proj']:
        projection = getattr(mlp, name)
        if getattr(projection.weight, 'quant_state', None) is not None:
            from bitsandbytes.functional import dequantize_4bit
            weight = dequantize_4bit(projection.weight.data, projection.weight.quant_state)
        else:
            weight = projection.weight.detach()
        weights.append(weight.to(dtype=torch.bfloat16))
    return DenseGatedMLP(*weights, mlp.act_fn).eval().requires_grad_(False)


def apply_physical_edit(dense, edit, target, partner, basis):
    """Return modified stored weights, using the released exact vectors where needed."""
    reference = {'gate_target': dense.gate_proj.weight[target],
                 'up_partner': dense.up_proj.weight[partner],
                 'down_target': dense.down_proj.weight[:, target]}
    for name, expected in reference.items():
        if not torch.equal(basis[name].to(expected.device, expected.dtype), expected):
            raise ValueError('Basis does not match the pinned dense target: ' + name)
    if edit == 'dense_delete':
        return delete_channels(dense, [target])
    if edit == 'dense_scale2':
        return edit_output_vectors(dense, [target], basis['down_target'].unsqueeze(1))
    if edit in ['dense_clone_unsplit', 'dense_clone_split']:
        return duplicate_channels(dense, [target], preserve_function=edit == 'dense_clone_split')
    if edit in ['dense_new_zero', 'dense_recombined']:
        down = basis['down_target'].unsqueeze(1)
        down = torch.zeros_like(down) if edit == 'dense_new_zero' else down / 2

def _manifest(rows):

def canonical(value):
    def check(item):
        if type(item) is dict:
            require(all(type(k) is str for k in item), 'JSON keys must be strings')

        def make(view):
            self._creator(view)
            require(view['pending'] is None, 'Pending work cannot be retried or skipped')
            ids = view['header']['contract']['measurement_ids']
            require(len(view['committed_ids']) < len(ids) and measurement_id == ids[len(view['committed_ids'])],
                    'Wrong, duplicate or reordered measurement ID')
            observed = _observation(state_reader, counter_reader)
            require(not observed['errors'], 'Actual starting observation unavailable')
            _state(observed['state']); _counters(observed['counters'])
            return {'id': measurement_id, 'context': context,
                    'before': {'state_sha256': digest(observed['state']), 'counters': observed['counters'],
                               'capture_time_ns': observed['time_ns']}}
        return self._append('intent', make)

    def prepare(self, result, checks, version_deltas, state_reader, counter_reader):
        result, checks, version_deltas = [json.loads(canonical(v)) for v in [result, checks, version_deltas]]
        require(type(result) is dict and type(checks) is dict and type(version_deltas) is dict,
                'Result, guards and versions require mappings')
        def make(view):
            self._creator(view)
            require(view['pending'] is not None and view['pending']['prepared'] is None, 'One pending intent required')
            observed = _observation(state_reader, counter_reader)
            require(not observed['errors'], 'Actual completion observation unavailable; stop as incomplete')
            _state(observed['state']); _counters(observed['counters'])
            new = dict(view['deltas'])
            for name, count in version_deltas.items():
                require(type(count) is int and count > 0, 'Invalid declared version delta')
                new[name] = new.get(name, 0) + count
            expected = _expected(view['header']['baseline']['state'], new)
            require(observed['state'] == expected, 'Observed state has not restored')
            return {'id': view['pending']['intent']['id'], 'result': result, 'checks': checks,
                    'version_deltas': version_deltas, 'counters': observed['counters'],
                    'state_witness': {'actual_state_sha256': digest(observed['state']),
                        'changed_parameter_versions': {n: observed['state']['parameters'][n]['version'] for n in new},
                        'capture_time_ns': observed['time_ns']}}
        return self._append('prepared', make)

    def commit(self):
        def make(view):
            pending = view['pending']
            require(pending is not None and pending['prepared'] is not None, 'Durable prepared evidence required')
            return {'id': pending['intent']['id'], 'prepared_digest': pending['prepared_digest']}
        return self._append('committed', make)

    def finish(self, state_reader, counter_reader):
        def make(view):
            self._creator(view)
            return {'observation': _observation(state_reader, counter_reader)}
        return self._append('complete', make)

    def stop(self, reason, state_reader=None, counter_reader=None):
        require(type(reason) is str and 0 < len(reason) <= 500, 'Bounded interruption reason required')
        def make(view):
            if state_reader is None and counter_reader is None:
                observation = {'state': None, 'counters': None,
                               'errors': {'state': 'Unavailable', 'counters': 'Unavailable'},
                               'pid': None, 'time_ns': time.time_ns()}
            else:
                self._creator(view)
                require(state_reader is not None and counter_reader is not None, 'Both observation callbacks required')
                observation = _observation(state_reader, counter_reader)
            return {'reason': reason, 'observation': observation,
                    'original_final_state_not_certified': True}
        return self._append('incomplete', make)
